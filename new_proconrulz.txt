############################
# JOIN / ROUND MESSAGES
############################
On Join;Say Welcome %p% to the server.
On Join;Admin;Say Admin [%p%] is online.
On Leave;Say %p% has left the server.
On Round;Say FHC BF3 server rules loaded. Type !rules or !menu for commands.
On Spawn;PlayerFirst;PlayerSay Type !rules for server rules and !menu for available commands.

############################
# BASIC PLAYER COMMANDS
############################
On Say;Text !rules;PlayerSay Rules: keep teams fair, no cheating, no abusive chat, follow admin decisions.
On Say;Text !rules;PlayerSay AutoAdmin may move high-impact players by KD/KPM and may kick suspicious KPM or headshot stats.
On Say;Text !menu;PlayerSay Commands: !rules !stats !killme !time !votekick @guid
On Say;Text !killme;Exec admin.killPlayer %p%
On Say;Text !suicide;Exec admin.killPlayer %p%
On Say;Text !time;PlayerSay Time: [%ymd%] [%hms%]
On Say;Text !alltime;Say Time Now: [%ymd%] [%hms%];Yell 8 Time Now: [%ymd%] [%hms%]
On Say;Text @guid;PlayerSay Your EA GUID: %ea_guid%

############################
# ADMIN COMMANDS
############################
On Say;Admin;Text !menu;PlayerSay Admin: !kill !kick !ban !tban !move1 !move2 !votecancel !nextlevel !yell !ptsay !ptyell
On Say;Admin;Text !admin;PlayerSay You are an admin. Admin-only commands are enabled.
On Say;Not Admin;Text !admin;PlayerSay You do not have administrator access.

On Say;Admin;Text !kill;TargetPlayer;Say Admin [%p%] killed %t%;TargetAction Kill 100
On Say;Admin;Text !kick;TargetPlayer;Say Admin [%p%] kicked %t%: %targettext%;TargetAction Kick Kicked by Admin %p% - %targettext%
On Say;Admin;Text !ban;TargetPlayer;Say Admin [%p%] banned %t%: %targettext%;TargetAction Ban Banned by Admin %p% - %targettext%
On Say;Admin;Text !tban;TargetPlayer;Say Admin [%p%] temp-banned %t%: %targettext%;TargetAction TempBan 1440 Banned by Admin %p% - %targettext%
On Say;Admin;Text !nextlevel;Exec admin.runNextLevel
On Say;Admin;Text !yell;Yell 10 %targettext%
On Say;Admin;Text !ptsay;TargetPlayer;TargetAction PlayerSay [ADMIN %p%] %targettext%
On Say;Admin;Text !ptyell;TargetPlayer;TargetAction PlayerYell 10 [ADMIN %p%] %targettext%
On Say;Admin;Text !move1;TargetPlayer;Say Admin [%p%] moved %t% to team 1;Exec admin.MovePlayer %t% 1 1 true
On Say;Admin;Text !move2;TargetPlayer;Say Admin [%p%] moved %t% to team 2;Exec admin.MovePlayer %t% 2 1 true

On Say;Not Admin;Text !kill;PlayerSay You do not have permission to use admin commands.
On Say;Not Admin;Text !kick;PlayerSay You do not have permission to use admin commands.
On Say;Not Admin;Text !ban;PlayerSay You do not have permission to use admin commands.
On Say;Not Admin;Text !tban;PlayerSay You do not have permission to use admin commands.

############################
# VOTE KICK
############################
# Vote kick is available only when no admin is online.
# Usage: !votekick <player> [reason], then other players type !votekick to vote yes.
On Round;Set %server_votekick_active% 0;Set %server_votekick_need% 5;Set %server_votekick_votes% 0;Set %server_votekick_id% 0
On Say;Admin;Text !votecancel;Set %server_votekick_active% 0;Set %server_votekick_votes% 0;Say Admin [%p%] cancelled the active vote kick.
On Say;Admins;Text !votekick;PlayerSay An admin is online. Please ask the admin to handle the issue.
On Say;Not Admins;Text !votekick;If %server_votekick_active% == 0;TargetPlayer;Incr %server_votekick_id%;Set %server_votekick_target% %t%;Set %server_votekick_reason% %targettext%;Set %server_votekick_active% 1;Set %server_votekick_votes% 1;Set %server_voted[%p%]% %server_votekick_id%;Say Vote kick started against %t% by %p%. Type !votekick to support. Need %server_votekick_need% votes.;Yell 10 Vote kick: %t%. Type !votekick to support.;End
On Say;Not Admins;Text !votekick;If %server_votekick_active% == 1;If %server_voted[%p%]% == %server_votekick_id%;PlayerSay You have already voted against %server_votekick_target%.;End
On Say;Not Admins;Text !votekick;If %server_votekick_active% == 1;If %server_voted[%p%]% != %server_votekick_id%;Incr %server_votekick_votes%;Set %server_voted[%p%]% %server_votekick_id%;Say Vote kick against %server_votekick_target%: %server_votekick_votes%/%server_votekick_need%.
On Say;Not Admins;Text !votekick;If %server_votekick_active% == 1;If %server_votekick_votes% >= %server_votekick_need%;TargetPlayer %server_votekick_target%;Say Vote kick passed against %t%.;TargetAction Kick Vote kick: %server_votekick_reason%;Set %server_votekick_active% 0;Set %server_votekick_votes% 0
On Leave;If %p% == %server_votekick_target%;Set %server_votekick_active% 0;Set %server_votekick_votes% 0;Say Vote kick target left the server. Vote cancelled.

############################
# ROUND STAT TRACKING
############################
On Join;Set %server_joinsec[%p%]% %seconds%;Set %server_kills[%p%]% 0;Set %server_deaths[%p%]% 0;Set %server_headshots[%p%]% 0
On Spawn;PlayerFirst;If %server_joinsec[%p%]% == 0;Set %server_joinsec[%p%]% %seconds%

On Kill;Incr %server_kills[%p%]%
On Kill;Set %server_deaths[%v%]% %server_deaths[%v%]% + 1
On Kill;Headshot;Incr %server_headshots[%p%]%
On Kill;Set %server_playsec[%p%]% %seconds% - %server_joinsec[%p%]%
On Kill;If %server_playsec[%p%]% < 60;Set %server_playsec[%p%]% 60
On Kill;Set %server_kpm.2[%p%]% %server_kills[%p%]% * 60 / %server_playsec[%p%]%
On Kill;If %server_deaths[%p%]% == 0;Set %server_kd.2[%p%]% %server_kills[%p%]%
On Kill;If %server_deaths[%p%]% > 0;Set %server_kd.2[%p%]% %server_kills[%p%]% / %server_deaths[%p%]%
On Kill;If %server_kills[%p%]% > 0;Set %server_hsrate.1[%p%]% %server_headshots[%p%]% * 100 / %server_kills[%p%]%
On Kill;Set %server_power.1[%p%]% %server_kd.2[%p%]% * 100 + %server_kpm.2[%p%]% * 100 + %server_kills[%p%]%

On Say;Text !stats;PlayerSay %p% round stats: K %server_kills[%p%]% / D %server_deaths[%p%]% / KD %server_kd.2[%p%]% / KPM %server_kpm.2[%p%]% / HS %server_hsrate.1[%p%]%%
On Kill;If %server_deaths[%p%]% == 0;VictimSay Killed by %p% | K %server_kills[%p%]% D %server_deaths[%p%]% KD %server_kills[%p%]% KPM %server_kpm.2[%p%]% HS %server_hsrate.1[%p%]%%
On Kill;If %server_deaths[%p%]% > 0;VictimSay Killed by %p% | K %server_kills[%p%]% D %server_deaths[%p%]% KD %server_kd.2[%p%]% KPM %server_kpm.2[%p%]% HS %server_hsrate.1[%p%]%%

############################
# KD / KPM TEAM BALANCE
############################
# Balance score uses current-round KD, KPM and kills:
#   power = KD*100 + KPM*100 + kills
# Move only strong-impact players, and only when their current team is larger or clearly ahead on tickets.
On Round;Set %server_balance_kills% 19;Set %server_balance_min_kills% 20;Set %server_balance_min_kd% 5;Set %server_balance_power% 520;Set %server_balance_ticketgap% 75
On Say;Admin;Text !balance_debug;Set %server_balance_kills% 19;Set %server_balance_min_kills% 20;Set %server_balance_min_kd% 5;Set %server_balance_power% 350;Set %server_balance_ticketgap% 25;PlayerSay Balance debug thresholds enabled.
On Say;Admin;Text !balance_normal;Set %server_balance_kills% 19;Set %server_balance_min_kills% 20;Set %server_balance_min_kd% 5;Set %server_balance_power% 520;Set %server_balance_ticketgap% 75;PlayerSay Balance normal thresholds restored.

On Kill;Not Admin;If %server_kills[%p%]% >= %server_balance_min_kills%;If %server_kd.2[%p%]% >= %server_balance_min_kd%;If %server_kills[%p%]% > %server_balance_kills%;If %server_power.1[%p%]% > %server_balance_power%;Set %server_balance_warn[%p%]% 1
On Kill;Not Admin;If %server_balance_warn[%p%]% == 1;PlayerSay [AUTOADMIN] You are a high-impact player this round. You may be moved for balance. KD %server_kd.2[%p%]% KPM %server_kpm.2[%p%]%.
On Kill;Not Admin;If %server_balance_warn[%p%]% == 1;Set %server_balance_warn[%p%]% 2

# Team 1 is larger than Team 2.
On Kill;Not Admin;Team 1;If %server_balance_warn[%p%]% == 2;If %server_kills[%p%]% >= %server_balance_min_kills%;If %server_kd.2[%p%]% >= %server_balance_min_kd%;If %server_kills[%p%]% > %server_balance_kills%;If %server_power.1[%p%]% > %server_balance_power%;If %ts1% > %ts2%;Exec admin.MovePlayer %p% 2 1 true;PlayerSay [AUTOADMIN] You were moved to balance teams by KD/KPM.

# Team 2 is larger than Team 1.
On Kill;Not Admin;Team 2;If %server_balance_warn[%p%]% == 2;If %server_kills[%p%]% >= %server_balance_min_kills%;If %server_kd.2[%p%]% >= %server_balance_min_kd%;If %server_kills[%p%]% > %server_balance_kills%;If %server_power.1[%p%]% > %server_balance_power%;If %ts2% > %ts1%;Exec admin.MovePlayer %p% 1 1 true;PlayerSay [AUTOADMIN] You were moved to balance teams by KD/KPM.

# Ticket advantage fallback when team sizes are equal or near equal.
On Kill;Not Admin;Team 1;If %server_balance_warn[%p%]% == 2;If %server_kills[%p%]% >= %server_balance_min_kills%;If %server_kd.2[%p%]% >= %server_balance_min_kd%;If %server_kills[%p%]% > %server_balance_kills%;If %server_power.1[%p%]% > %server_balance_power%;If %team_score% > %server_team_score[2]% + %server_balance_ticketgap%;Exec admin.MovePlayer %p% 2 1 true;PlayerSay [AUTOADMIN] You were moved because your team is ahead and your KD/KPM is high.
On Kill;Not Admin;Team 2;If %server_balance_warn[%p%]% == 2;If %server_kills[%p%]% >= %server_balance_min_kills%;If %server_kd.2[%p%]% >= %server_balance_min_kd%;If %server_kills[%p%]% > %server_balance_kills%;If %server_power.1[%p%]% > %server_balance_power%;If %team_score% > %server_team_score[1]% + %server_balance_ticketgap%;Exec admin.MovePlayer %p% 1 1 true;PlayerSay [AUTOADMIN] You were moved because your team is ahead and your KD/KPM is high.

############################
# SUSPICIOUS KPM / HEADSHOT CONTROL
############################
# KPM thresholds are intentionally high for BF3 public play. Admins are excluded.
# Warnings accumulate per round. At 3 warnings the player is kicked for review.
On Round;Set %server_warn_kpm% 3.5;Set %server_kick_kpm% 5.0;Set %server_warn_hs% 75;Set %server_kick_hs% 85
On Say;Admin;Text !anti_debug;Set %server_warn_kpm% 2.5;Set %server_kick_kpm% 3.5;Set %server_warn_hs% 65;Set %server_kick_hs% 75;PlayerSay Anti-cheat debug thresholds enabled.
On Say;Admin;Text !anti_normal;Set %server_warn_kpm% 3.5;Set %server_kick_kpm% 5.0;Set %server_warn_hs% 75;Set %server_kick_hs% 85;PlayerSay Anti-cheat normal thresholds restored.

On Kill;Not Admin;If %server_kills[%p%]% > 12;If %server_playsec[%p%]% > 180;If %server_kpm.2[%p%]% > %server_warn_kpm%;Incr %server_suspect[%p%]%;PlayerYell 10 [AUTOADMIN] Suspicious KPM warning %server_suspect[%p%]%/3. Current KPM %server_kpm.2[%p%]%.
On Kill;Not Admin;If %server_kills[%p%]% > 12;If %server_playsec[%p%]% > 180;If %server_kpm.2[%p%]% > %server_kick_kpm%;Incr %server_suspect[%p%]%;PlayerYell 10 [AUTOADMIN] Extreme KPM warning %server_suspect[%p%]%/3. Current KPM %server_kpm.2[%p%]%.
On Kill;Not Admin;If %server_kills[%p%]% > 20;If %server_hsrate.1[%p%]% > %server_warn_hs%;Incr %server_suspect[%p%]%;PlayerYell 10 [AUTOADMIN] Suspicious headshot rate warning %server_suspect[%p%]%/3. HS %server_hsrate.1[%p%]%%.
On Kill;Not Admin;If %server_kills[%p%]% > 30;If %server_hsrate.1[%p%]% > %server_kick_hs%;Incr %server_suspect[%p%]%;PlayerYell 10 [AUTOADMIN] Extreme headshot rate warning %server_suspect[%p%]%/3. HS %server_hsrate.1[%p%]%%.
On Kill;Not Admin;If %server_suspect[%p%]% > 2;Say [AUTOADMIN] %p% kicked for suspicious KPM/HS stats. KPM %server_kpm.2[%p%]% HS %server_hsrate.1[%p%]%%;Kick Suspicious KPM or headshot rate;Set %server_suspect[%p%]% 0

# MAP / WEAPON LIMITS
############################
# Operation Metro explosive limiter, no privilege bypass except Admin.
On Round;Map MP_Subway;Set %server_no_explosive% 1
On Round;Not Map MP_Subway;Set %server_no_explosive% 0
On RoundOver;Set %server_no_explosive% 0
On Say;Admin;Text !ban_expl;Set %server_no_explosive% 1;Say Admin [%p%] disabled explosive kills.;Yell 10 Explosive kills are disabled.
On Say;Admin;Text !ok_expl;Set %server_no_explosive% 0;Say Admin [%p%] enabled explosive kills.;Yell 10 Explosive kills are enabled.
On Say;Admin;Text !check_expl;PlayerSay Explosive limit value: %server_no_explosive%

On Kill;Not Admin;Damage ProjectileExplosive;If %server_no_explosive% == 1;PlayerFirst;Set %server_explosivecnt[%p%]% 1;Kill;PlayerYell 10 No explosive kills on this map.
On Kill;Not Admin;Damage ProjectileExplosive;If %server_no_explosive% == 1;If %server_explosivecnt[%p%]% > 0;If %server_explosivecnt[%p%]% < 3;Incr %server_explosivecnt[%p%]%;Kill;PlayerYell 10 No explosive kills. Warning %server_explosivecnt[%p%]%/3.
On Kill;Not Admin;Damage ProjectileExplosive;If %server_no_explosive% == 1;If %server_explosivecnt[%p%]% > 2;Say [AUTOADMIN] %p% kicked for repeated explosive kills.;Kick No explosive kills;Set %server_explosivecnt[%p%]% 0
